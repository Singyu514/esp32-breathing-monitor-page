<script>

const SERVICE_UUID = "f0a00001-7c5a-4a26-9e3d-6c82f0a19b21";

const STATE_UUID = "f0a00002-7c5a-4a26-9e3d-6c82f0a19b21";


const STATES = {

  CALM: { name: "CALM BREATHING", color: "#35d399" },

  FAST: { name: "FAST BREATHING", color: "#ff5964" },

  NOT_ATTACHED: { name: "NOT ATTACHED", color: "#58a6ff" },

  SHAKING: { name: "SHAKING", color: "#d88aff" }

};


let device = null;

let characteristic = null;

let currentState = null;

let stateSince = null;

let history = [];

let mode = "live";

let connectedOnce = false;

let unexpectedDisconnect = false;


const el = id => document.getElementById(id);


function log(message) {

  const row = document.createElement("div");

  row.className = "log-entry";

  row.textContent = new Date().toLocaleTimeString() + " — " + message;

  el("log").prepend(row);

}


function setState(raw) {

  const state = raw.trim().toUpperCase();

  if (!STATES[state]) {

    log("Unknown state received: " + raw);

    return;

  }


  if (state !== currentState) {

    currentState = state;

    stateSince = Date.now();

    log("State changed: " + STATES[state].name);

  }


  const info = STATES[state];

  el("stateName").textContent = info.name;

  el("stateName").style.color = info.color;

  el("indicator").style.background = info.color;

}


function onNotification(event) {

  if (mode !== "live") return;

  const value = event.target.value;

  const bytes = new Uint8Array(

    value.buffer, value.byteOffset, value.byteLength

  );

  const state = new TextDecoder("utf-8").decode(bytes);

  setState(state);

}


async function connectBLE() {

  if (!navigator.bluetooth) {

    alert("Web Bluetooth unavailable. Use Chrome on macOS, Windows or Android over HTTPS.");

    return;

  }


  try {

    mode = "live";

    el("modeLabel").textContent = "Mode: Live BLE";


    device = await navigator.bluetooth.requestDevice({

      filters: [{ services: [SERVICE_UUID] }],

      optionalServices: [SERVICE_UUID]

    });


    device.addEventListener("gattserverdisconnected", onDisconnect);


    const server = await device.gatt.connect();

    const service = await server.getPrimaryService(SERVICE_UUID);

    characteristic = await service.getCharacteristic(STATE_UUID);


    await characteristic.startNotifications();

    characteristic.addEventListener(

      "characteristicvaluechanged", onNotification

    );


    connectedOnce = true;

    unexpectedDisconnect = false;


    el("connection").textContent = "● Connected";

    el("connection").style.color = "#35d399";

    el("connectionMessage").textContent = "Connected";

    log("BLE connected");


    try {

      const value = await characteristic.readValue();

      onNotification({ target: { value } });

    } catch (error) {

      log("Waiting for first state notification");

    }

  } catch (error) {

    log("BLE connection error: " + error.message);

    el("connection").textContent = "● Disconnected";

  }

}


function onDisconnect() {

  if (!connectedOnce) return;


  el("connection").textContent = "● Connection lost";

  el("connection").style.color = "#ff5964";


  if (mode === "live") {

    unexpectedDisconnect = true;

    currentState = null;

    stateSince = null;

    log("ALARM: BLE connection lost");

  }

}


function disconnectBLE() {

  unexpectedDisconnect = false;

  if (device && device.gatt.connected) {

    device.gatt.disconnect();

  }

  currentState = null;

  stateSince = null;

  el("connection").textContent = "● Disconnected";

  el("connection").style.color = "#9aabc3";

  log("Disconnected by user");

}


function demo(state) {

  if (device && device.gatt.connected) {

    disconnectBLE();

  }

  mode = "demo";

  unexpectedDisconnect = false;

  el("modeLabel").textContent = "Mode: DEMO — Simulated Data";

  setState(state);

}


function stopDemo() {

  mode = "live";

  currentState = null;

  stateSince = null;

  el("modeLabel").textContent = "Mode: Live BLE";

  el("stateName").textContent = "WAITING FOR DATA";

  el("stateName").style.color = "#eef4ff";

  el("indicator").style.background = "#64748b";

  log("Demo stopped");

}


function setAlarm(id, messageId, active, message) {

  el(id).classList.toggle("active", active);

  el(messageId).textContent = message;

}


function tick() {

  const now = Date.now();

  const seconds = stateSince

    ? Math.floor((now - stateSince) / 1000)

    : 0;


  el("stateDuration").textContent = currentState

    ? "Current state: " + seconds + " seconds"

    : "No active state";


  const fastLimit = Math.max(1, Number(el("fastLimit").value) || 10);

  const offLimit = Math.max(1, Number(el("offLimit").value) || 15);


  const fastActive = currentState === "FAST" && seconds >= fastLimit;

  const offActive =

    currentState === "NOT_ATTACHED" && seconds >= offLimit;


  setAlarm(

    "fastAlarm", "fastMessage", fastActive,

    fastActive ? "ALARM: Fast breathing too long!" : "Normal"

  );


  setAlarm(

    "offAlarm", "offMessage", offActive,

    offActive ? "ALARM: Sensor not attached!" : "Normal"

  );


  setAlarm(

    "connectionAlarm", "connectionMessage",

    unexpectedDisconnect,

    unexpectedDisconnect ? "ALARM: BLE connection lost!" :

    (device && device.gatt.connected ? "Connected" : "Not connected")

  );


  history.push(currentState);

  if (history.length > 120) history.shift();


  const timeline = el("timeline");

  timeline.replaceChildren();


  for (const state of history) {

    const segment = document.createElement("div");

    segment.className = "segment";

    segment.style.flex = "1";

    segment.style.background = STATES[state]?.color || "#334155";

    timeline.appendChild(segment);

  }

}


el("connectBtn").addEventListener("click", connectBLE);

el("disconnectBtn").addEventListener("click", disconnectBLE);


setInterval(tick, 1000);

log("Dashboard ready");

</script>
